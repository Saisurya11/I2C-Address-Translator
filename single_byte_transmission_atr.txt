`define N 3
module i2c_atr(sda_in,scl_in,sda_out,scl_out,en,ready,translation_byte);
//parameter `ports=3;

inout sda_in;
inout scl_in;

inout [`N-1:0] sda_out,scl_out;

output [`N-1:0]ready;

input [`N-1:0]en;

input [6:0]translation_byte;

wire [7:0] byte_buffer[`N-1:0];

wire scl_0;
assign scl_0=scl_in;
integer count=0;

reg flag=0;

reg r_st_bit_current,r_st_bit_prev;

always@(sda_in, r_st_bit_current) begin
r_st_bit_current<=sda_in;
r_st_bit_prev<=r_st_bit_current;
end

wire st_bit;
assign st_bit=(sda_in==0 && scl_in && count==0)?1'b1:1'b0;

wire stop_bit=(sda_in==1 && scl_in && count==0)?1'b1:1'b0;

wire r_st=(r_st_bit_prev==1 && sda_in==0 && flag==1 && count==19)?1'b1:1'b0;

wire [`N-1:0]enables;
assign enables=en;

integer number;
integer en_i;
always@(*) begin
if(st_bit) begin
flag=1;
for(en_i=0;en_i<`N;en_i=en_i+1) begin
if(enables[en_i]) number=en_i;
end
end
else if(stop_bit)
flag=0;
end

genvar i;
generate
for(i=0;i<`N;i=i+1) begin
assign ready[i]=(en[i]==1)?1'b1:1'b0;
assign byte_buffer[i][6:0]=(en[i]==1)?translation_byte:0;
end
endgenerate

reg r_st_flag;
always@(*) begin
if(flag==1 && count==19 && r_st==1)
r_st_flag=1'b1;
else if(stop_bit)
r_st_flag=1'b0;
end

genvar j;
generate
for(j=0;j<`N;j=j+1) begin

wire step_1=(flag==1 && number==j && count>=1 && count<=7)?1'b1:1'b0; //perofrms xor
wire w_r_1=(flag==1 && number==j && count==8)?1'b1:1'b0; //not perform xor and send write_bit 
wire mem_address =(flag==1 && number==j && count>=10 && count<=17)?1'b1:1'b0; //sending the memory address for the slave
wire slave_address_r_st= (flag==1 && number==j && count>=20 && count<=26 && r_st_flag); //sending the slave address after the r_start
wire sending_wr_data= (flag==1 && number==j && count>=20 && count<=26 && r_st_flag==0); //sedning the write data in the normal condition
wire read_bit_and_master_ack=flag==1 && number==j && (count==27 || count==37)&& r_st_flag;  //capturing the read bit and also getting the ack from the master after reading
wire default_value_en_applied=(number!=j || count==0) && en[j]==1 ;  //sending the defualt values for the slaves during normal time when no transactions happening
wire ack_release_and_read_window= ((number==j && count%9==0 && count>=1 && r_st_flag==0) || (count==28 && number==j && r_st_flag) || (count>=29 && count<=36 && number==j));
wire initial_and_r_st=((flag==1 && number==j &&(count==0 || count==19))); //sending the defualt value to the slave after the atr starts, before we ill drive one, becuase by that time number not detected as it is in always@(*) block

assign sda_out[j]=(step_1)?(sda_in^byte_buffer[number][7-count]):
(w_r_1 || mem_address || initial_and_r_st || sending_wr_data || read_bit_and_master_ack)?sda_in:
(slave_address_r_st)?(sda_in^byte_buffer[number][26-count]):
(default_value_en_applied)?sda_in:(ack_release_and_read_window)?1'bz:(en[j]==1 && number==='dx && count==0)?1'b1:1'bz;

assign scl_out[j]=(flag==1 && j==number)?scl_in:1'b1;
end
endgenerate

assign sda_in=((count%9==0 && count>=1 && flag==1 && r_st_flag==0) ||((count>=28 && count<=36 ) && r_st_flag))?sda_out[number]:1'bz;

always@(negedge scl_0) begin
count<=count+1;

if((count==27 && r_st_flag==0) || (r_st_flag==1 && count==37))
count<=0;
end
endmodule
